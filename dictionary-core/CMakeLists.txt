cmake_minimum_required(VERSION 3.22)
project(dictionary_core LANGUAGES C CXX)

set(RAPIDJSON_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tools/native-dictionary/.deps/rapidjson/include" CACHE PATH "Directory containing rapidjson/reader.h")
if(NOT EXISTS "${RAPIDJSON_INCLUDE_DIR}/rapidjson/reader.h")
    message(FATAL_ERROR "Set RAPIDJSON_INCLUDE_DIR to the RapidJSON include directory")
endif()
set(LEXBOR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tools/native-dictionary/.deps/lexbor" CACHE PATH "Lexbor source checkout")
if(NOT EXISTS "${LEXBOR_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set LEXBOR_SOURCE_DIR to the Lexbor checkout")
endif()
set(LEXBOR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LEXBOR_SOURCE_DIR}" lexbor EXCLUDE_FROM_ALL)
set_target_properties(lexbor_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(dictionary_core SHARED src/dictionary_core.cpp src/styled_document.cpp src/document_layout.cpp src/drawing_plan.cpp)
target_compile_features(dictionary_core PUBLIC cxx_std_20)
target_include_directories(dictionary_core PUBLIC include PRIVATE src "${RAPIDJSON_INCLUDE_DIR}")
target_compile_definitions(dictionary_core PRIVATE DC_BUILD)
target_include_directories(dictionary_core PRIVATE "${LEXBOR_SOURCE_DIR}/source")
target_link_libraries(dictionary_core PRIVATE lexbor_static)
if(ANDROID)
    target_compile_options(dictionary_core PRIVATE -O3 -ffunction-sections -fdata-sections)
    target_link_options(dictionary_core PRIVATE -Wl,--gc-sections)
endif()
set_target_properties(dictionary_core PROPERTIES CXX_VISIBILITY_PRESET hidden)

option(DC_BUILD_BENCHMARK "Build the standalone captured-source benchmark" ON)
if(DC_BUILD_BENCHMARK)
    add_executable(dictionary_styled_bench ../tools/native-dictionary/styled_benchmark.cpp)
    target_compile_features(dictionary_styled_bench PRIVATE cxx_std_20)
    target_include_directories(dictionary_styled_bench PRIVATE "${RAPIDJSON_INCLUDE_DIR}")
    target_link_libraries(dictionary_styled_bench PRIVATE dictionary_core)
    if(WIN32)
        target_sources(dictionary_styled_bench PRIVATE ../tools/native-dictionary/windows_text_engine.cpp)
        target_link_libraries(dictionary_styled_bench PRIVATE dwrite)
        target_compile_definitions(dictionary_styled_bench PRIVATE DC_WINDOWS_TEXT_ENGINE)
    endif()
    add_executable(dictionary_core_bench ../tools/native-dictionary/benchmark.cpp)
    target_compile_features(dictionary_core_bench PRIVATE cxx_std_23)
    target_include_directories(dictionary_core_bench PRIVATE include src "${RAPIDJSON_INCLUDE_DIR}")
    target_include_directories(dictionary_core_bench PRIVATE adapters)
    if(DEFINED HOSHIDICTS_INCLUDE_DIR)
        target_include_directories(dictionary_core_bench PRIVATE "${HOSHIDICTS_INCLUDE_DIR}")
        target_compile_definitions(dictionary_core_bench PRIVATE DC_WITH_HOSHI_ADAPTER)
    endif()
    target_link_libraries(dictionary_core_bench PRIVATE dictionary_core)
    if(DEFINED GLAZE_INCLUDE_DIR)
        target_include_directories(dictionary_core_bench PRIVATE "${GLAZE_INCLUDE_DIR}")
        target_compile_definitions(dictionary_core_bench PRIVATE DC_WITH_GLAZE)
    endif()
endif()
